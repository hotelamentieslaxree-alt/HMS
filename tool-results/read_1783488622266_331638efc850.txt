    27→export function DashboardModule() {
    28→  const { role } = useAppStore();
    29→
    30→  // Route to the correct dashboard based on role
    31→  const roleLevel = role.includes("owner") ? "owner" :
    32→    role === "gm" ? "gm" :
    33→    role === "fom" || role === "receptionist" ? "frontdesk" :
    34→    role === "hk_mgr" || role === "hk_attendant" ? "housekeeping" :
    35→    role === "fb_mgr" || role === "waiter" ? "fnb" :
    36→    role === "fin_mgr" ? "finance" :
    37→    role === "eng_mgr" || role === "technician" ? "engineering" :
    38→    role === "rev_mgr" ? "revenue" :
    39→    role === "hr_mgr" ? "hr" :
    40→    role === "sales_mgr" || role === "sales_exec" ? "sales" :
    41→    role === "mkt_mgr" || role === "mkt_exec" ? "marketing" :
    42→    "gm"; // fallback
    43→
    44→  switch (roleLevel) {
    45→    case "owner": return <OwnerDashboard />;
    46→    case "gm": return <GMDashboard />;
    47→    case "frontdesk": return <FrontDeskDashboard />;
    48→    case "housekeeping": return <HousekeepingDashboard />;
    49→    case "fnb": return <FnBDashboard />;
    50→    case "finance": return <FinanceDashboard />;
    51→    case "engineering": return <EngineeringDashboard />;
    52→    case "revenue": return <RevenueDashboard />;
    53→    case "hr": return <HRDashboard />;
    54→    case "sales": return <SalesDashboard />;
    55→    case "marketing": return <MarketingDashboard />;
    56→    default: return <GMDashboard />;
    57→  }
    58→}

    60→// ─── Shared hook for dashboard data ─────────────────────────────
    61→function useDashboardData() {
    62→  const { data, loading, reload } = useApi<any>("/api/dashboard", []);
    63→  const { refreshTick } = useAppStore();
    64→  useEffect(() => {
    65→    if (refreshTick > 0) reload();
    66→  }, [refreshTick, reload]);
    67→  return { data, loading };
    68→}
