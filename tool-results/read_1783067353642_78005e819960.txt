   137→function OverviewTab() {
   138→  const { data: empData, loading: l1 } = useApi<any>("/api/hr/employees?isActive=all", []);
   139→  const { data: attData, loading: l2 } = useApi<any>(`/api/hr/attendance?month=${now.getMonth()+1}&year=${now.getFullYear()}`, []);
   140→  const { data: payrollData, loading: l3 } = useApi<any>(`/api/hr/payroll?month=${now.getMonth()+1}&year=${now.getFullYear()}`, []);
   141→  const { data: eventData, loading: l4 } = useApi<any>(`/api/hr/events?month=${now.getMonth()+1}&year=${now.getFullYear()}`, []);
   142→
   143→  const loading = l1 || l2 || l3 || l4;
   144→
   145→  if (loading || !empData) return <SkeletonGrid />;
   146→
   147→  const emps: any[] = Array.isArray(empData) ? empData : [];
   148→  const totalEmployees = emps.length;
   149→  const activeCount = emps.filter((e: any) => e.isActive).length;
   150→  const byDepartment: Record<string, number> = {};
   151→  for (const e of emps) {
   152→    const d = e.department || "Unassigned";
   153→    byDepartment[d] = (byDepartment[d] || 0) + 1;
   154→  }
   155→
   156→  const attSummary = (attData as any)?.summary || {};
   157→  const payrollSummary = (payrollData as any)?.summary || {};
   158→  const events: any[] = (eventData as any)?.events || [];
   159→
   160→  // Upcoming birthdays this/next month
   161→  const thisMonth = now.getMonth() + 1;
   162→  const nextMonth = thisMonth === 12 ? 1 : thisMonth + 1;
   163→  const upcomingBdays = MOCK_BIRTHDAYS.filter(b => b.birthMonth === thisMonth || b.birthMonth === nextMonth)
   164→    .sort((a, b) => {
   165→      if (a.birthMonth !== b.birthMonth) return a.birthMonth - b.birthMonth;
   166→      return a.birthDay - b.birthDay;
   167→    });
   168→
   169→  // Attendance trend (last 7 days)
   170→  const attRecords: any[] = (attData as any)?.attendance || [];
   171→  const trend: { name: string; Present: number; Absent: number }[] = [];
   172→  for (let i = 6; i >= 0; i--) {
   173→    const d = new Date(); d.setDate(d.getDate() - i);
   174→    const ds = d.toISOString().slice(0, 10);
   175→    const dayRecs = attRecords.filter((r: any) => new Date(r.date).toISOString().slice(0, 10) === ds);
   176→    trend.push({
   177→      name: d.toLocaleDateString("en-IN", { weekday: "short" }),
   178→      Present: dayRecs.filter((r: any) => r.status === "present" || r.status === "late").length,
   179→      Absent: dayRecs.filter((r: any) => r.status === "absent").length,
   180→    });
   181→  }

   183→  const deptPie = Object.entries(byDepartment).map(([name, value]) => ({ name, value: value as number }));
   184→

   186→  const avgSalary = 45000;
   187→  const estPayroll = (payrollSummary.totalNetPay as number) || activeCount * avgSalary;

   189→  return (

   191→      {/* KPI Row */}
   192→      <div className="grid grid-cols-2 md:grid-cols-3 xl:grid-cols-6 gap-3">

   197→        <KpiCard label="Upcoming Birthdays" value={upcomingBdays.length} icon={Cake} accent="error" hint="This & next month" />
   198→        <KpiCard label="Avg Scorecard" value={78.5} icon={Star} accent="warning" />
   199→      </div>
   200→
   201→      {/* Charts Row */}
   202→      <div className="grid grid-cols-1 lg:grid-cols-2 gap-4">
   203→        <Card>
   204→          <CardHeader className="pb-2"><CardTitle className="font-display text-sm">Attendance Trend (7 days)</CardTitle></CardHeader>
   205→          <CardContent>
   206→            <ResponsiveContainer width="100%" height={220}>
   207→              <BarChart data={trend}>
   208→                <CartesianGrid strokeDasharray="3 3" />

   211→                <Tooltip />

   334→function EmployeesTab() {
   335→  const [search, setSearch] = useState("");
   336→  const [deptFilter, setDeptFilter] = useState("all");
   337→  const [roleFilter, setRoleFilter] = useState("all");
   338→  const [statusFilter, setStatusFilter] = useState("all");
   339→  const [addOpen, setAddOpen] = useState(false);
   340→  const [editItem, setEditItem] = useState<any>(null);
   341→  const [form, setForm] = useState({
   342→    firstName: "", lastName: "", email: "", role: "", department: "", phone: "",
   343→  });
   344→
   345→  const queryParams = new URLSearchParams({ isActive: statusFilter });
   346→  if (search) queryParams.set("search", search);
   347→  if (deptFilter !== "all") queryParams.set("department", deptFilter);
   348→  if (roleFilter !== "all") queryParams.set("role", roleFilter);
   349→
   350→  const { data, loading, reload } = useApi<any>(`/api/hr/employees?${queryParams.toString()}`, [search, deptFilter, roleFilter, statusFilter]);
   351→  const { triggerRefresh } = useAppStore();
   352→
   353→  const employees: any[] = Array.isArray(data) ? data : [];
   354→  const summary = (data as any)?.summary || {};
   355→  const departments: string[] = Object.keys(summary.byDepartment || {});
   356→
   357→  const resetForm = () => setForm({ firstName: "", lastName: "", email: "", role: "", department: "", phone: "" });
   358→
   359→  const handleSave = async () => {
   360→    try {
   361→      const body = { ...form, departmentId: form.department || undefined };
   362→      if (editItem) {
   363→        await apiPut("/api/hr/employees", { id: editItem.id, ...body });
   364→        toast.success("Employee updated");
   365→      } else {
   366→        await apiPost("/api/hr/employees", body);
   367→        toast.success("Employee added");
   368→      }
   369→      setAddOpen(false); setEditItem(null); resetForm(); reload(); triggerRefresh();
   370→    } catch (e: any) { toast.error(e.message); }
   371→  };
   372→
   373→  const handleDelete = async (id: string) => {
   374→    try {
   375→      await apiPut("/api/hr/employees", { id, isActive: false });
   376→      toast.success("Employee deactivated"); reload(); triggerRefresh();
   377→    } catch (e: any) { toast.error(e.message); }
   378→  };
   379→
   380→  if (loading || !data) return <SkeletonGrid />;
   381→
   382→  return (
   383→    <div className="space-y-4">
   384→      {/* Filters */}
   385→      <div className="flex flex-wrap items-center gap-2">
   386→        <div className="relative flex-1 min-w-[200px]">
   387→          <Search className="absolute left-2.5 top-2.5 h-4 w-4 text-muted-foreground" />
   388→          <Input className="pl-9" placeholder="Search employees..." value={search} onChange={e => setSearch(e.target.value)} />
   389→        </div>

   782→      setForm({ title: "", type: "meeting", eventDate: "", endDate: "", venue: "", description: "" });
   783→      reload(); triggerRefresh();

   785→  };

   788→    try {
   789→      await apiPut("/api/hr/events", { id, status: "cancelled" });
   790→      toast.success("Event cancelled"); reload(); triggerRefresh();
   791→    } catch (e: any) { toast.error(e.message); }
   792→  };
   793→
   794→  const handleCreateBirthdayEvent = (b: typeof MOCK_BIRTHDAYS[0]) => {
   795→    const bDate = new Date(now.getFullYear(), b.birthMonth - 1, b.birthDay);
   796→    // If birthday already passed this year, use next year
   797→    if (bDate < now) bDate.setFullYear(bDate.getFullYear() + 1);
   798→    setBirthdayForm({

   800→      birthDay: String(b.birthDay),

   977→                          endDate: e.endDate?.slice(0,10) || "", venue: e.venue || "", description: e.description || "",

   980→                        setAddOpen(true);

   991→            })}

   996→      {/* Add/Edit Event Dialog */}
   997→      <Dialog open={addOpen} onOpenChange={setAddOpen}>
