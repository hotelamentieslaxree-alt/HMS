    92→// ─── HELPERS ────────────────────────────────────────────────────────
    93→function getInitials(firstName?: string, lastName?: string): string {
    94→  const f = firstName?.trim()?.[0] ?? "";
    95→  const l = lastName?.trim()?.[0] ?? "";
    96→  return (f + l).toUpperCase() || "?";
    97→}
    98→
    99→function getDepartment(emp: any): string {
   100→  if (emp.department?.name) return emp.department.name;
   101→  if (emp.departmentCode) return emp.departmentCode;
   102→  return ROLE_DEPT[emp.role] || "Unassigned";
   103→}
   104→
   105→function getRoleLabel(role: string): string {
   106→  return ROLE_LABEL[role] || role?.replace(/_/g, " ").replace(/\b\w/g, (c) => c.toUpperCase()) || "Staff";
   107→}
   108→
   109→function getStatus(emp: any): string {
   110→  if (emp.isActive === false) return "inactive";
   111→  if (emp.isOnLeave) return "on_leave";
   112→  return "active";
   113→}

   217→// ─── TAB 1: DIRECTORY ──────────────────────────────────────────────
   218→function DirectoryTab() {
   219→  const { data: empData, loading, error } = useApi<any>("/api/hr/employees?isActive=all", []);
   220→  const [search, setSearch] = useState("");
   221→  const [deptFilter, setDeptFilter] = useState("all");
   222→  const [statusFilter, setStatusFilter] = useState("all");
   223→  const [selectedEmp, setSelectedEmp] = useState<any | null>(null);
   224→  const [detailOpen, setDetailOpen] = useState(false);
   225→
   226→  const employees: any[] = useMemo(() => {
   227→    if (!empData) return [];
   228→    return Array.isArray(empData) ? empData : [];
   229→  }, [empData]);
   230→
   231→  // Derive department list from data
   232→  const departments = useMemo(() => {
   233→    const set = new Set<string>();
   234→    employees.forEach((e) => set.add(getDepartment(e)));
   235→    return Array.from(set).sort();
   236→  }, [employees]);
   237→
   238→  // Filtered list
   239→  const filtered = useMemo(() => {
   240→    let list = [...employees];
   241→    if (search.trim()) {
   242→      const q = search.toLowerCase();
   243→      list = list.filter(
   244→        (e) =>
   245→          `${e.firstName} ${e.lastName}`.toLowerCase().includes(q) ||
   246→          (e.email || "").toLowerCase().includes(q) ||
   247→          (e.phone || "").includes(q) ||
   248→          (e.employeeCode || "").toLowerCase().includes(q) ||
   249→          getRoleLabel(e.role).toLowerCase().includes(q) ||
   250→          getDepartment(e).toLowerCase().includes(q)
   251→      );
   252→    }
   253→    if (deptFilter !== "all") {
   254→      list = list.filter((e) => getDepartment(e) === deptFilter);
   255→    }
   256→    if (statusFilter !== "all") {
   257→      if (statusFilter === "active") list = list.filter((e) => getStatus(e) === "active");
   258→      else if (statusFilter === "inactive") list = list.filter((e) => getStatus(e) === "inactive");
   259→      else if (statusFilter === "on_leave") list = list.filter((e) => getStatus(e) === "on_leave");
   260→    }
   261→    // Sort: active first, then by role level, then name
   262→    list.sort((a, b) => {
   263→      const sa = getStatus(a) === "active" ? 0 : getStatus(a) === "on_leave" ? 1 : 2;
   264→      const sb = getStatus(b) === "active" ? 0 : getStatus(b) === "on_leave" ? 1 : 2;
   265→      if (sa !== sb) return sa - sb;
   266→      const la = ROLE_LEVEL[a.role] ?? 9;
   267→      const lb = ROLE_LEVEL[b.role] ?? 9;
   268→      if (la !== lb) return la - lb;
   269→      return `${a.firstName} ${a.lastName}`.localeCompare(`${b.firstName} ${b.lastName}`);
   270→    });
   271→    return list;
   272→  }, [employees, search, deptFilter, statusFilter]);
   273→
   274→  // KPI stats
   275→  const totalStaff = employees.length;
   276→  const activeStaff = employees.filter((e) => getStatus(e) === "active").length;
   277→  const onLeave = employees.filter((e) => getStatus(e) === "on_leave").length;
   278→  const deptCount = departments.length;
   279→
   280→  const openDetail = (emp: any) => {
   281→    setSelectedEmp(emp);
   282→    setDetailOpen(true);
   283→  };
   284→
   285→  if (loading) return <SkeletonGrid />;
   286→
   287→  return (
   288→    <div className="space-y-5">
   289→      {/* KPI Cards */}
   290→      <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   291→        <KpiCard label="Total Staff" value={totalStaff} icon={Users} accent="navy" />
   292→        <KpiCard label="Active" value={activeStaff} icon={UserCheck} accent="success" />
   293→        <KpiCard label="On Leave" value={onLeave} icon={MapPin} accent="warning" />
   294→        <KpiCard label="Departments" value={deptCount} icon={Building2} accent="gold" />
   295→      </div>
   296→
   297→      {/* Search & Filters */}
   298→      <Card>
   299→        <CardContent className="p-4">
   300→          <div className="flex flex-col sm:flex-row gap-3">
   301→            <div className="relative flex-1">
   302→              <Search className="absolute left-3 top-1/2 h-4 w-4 -translate-y-1/2 text-muted-foreground" />
   303→              <Input
   304→                placeholder="Search by name, email, role, department..."
   305→                className="pl-9"
   306→                value={search}
   307→                onChange={(e) => setSearch(e.target.value)}
   308→              />
   309→            </div>
   310→            <Select value={deptFilter} onValueChange={setDeptFilter}>
   311→              <SelectTrigger className="w-full sm:w-[180px]">
   312→                <Building2 className="h-4 w-4 mr-2 text-muted-foreground" />
   313→                <SelectValue placeholder="Department" />
   314→              </SelectTrigger>
   315→              <SelectContent>
   316→                <SelectItem value="all">All Departments</SelectItem>
   317→                {departments.map((d) => (
   318→                  <SelectItem key={d} value={d}>{d}</SelectItem>
   319→                ))}
   320→              </SelectContent>
   321→            </Select>
   322→            <Select value={statusFilter} onValueChange={setStatusFilter}>
   323→              <SelectTrigger className="w-full sm:w-[140px]">
   324→                <Filter className="h-4 w-4 mr-2 text-muted-foreground" />
   325→                <SelectValue placeholder="Status" />
   326→              </SelectTrigger>
   327→              <SelectContent>
   328→                <SelectItem value="all">All Status</SelectItem>
   329→                <SelectItem value="active">Active</SelectItem>
   330→                <SelectItem value="on_leave">On Leave</SelectItem>
   331→                <SelectItem value="inactive">Inactive</SelectItem>
   332→              </SelectContent>
   333→            </Select>
   334→          </div>
   335→        </CardContent>
   336→      </Card>
   337→
   338→      {/* Staff Table */}
   339→      <Card>
   340→        <CardHeader className="pb-3">
   341→          <CardTitle className="font-display text-base">Staff Directory</CardTitle>
   342→          <CardDescription className="text-xs">
   343→            Showing {filtered.length} of {totalStaff} employees
   344→          </CardDescription>
   345→        </CardHeader>
   346→        <CardContent className="p-0">
   347→          {filtered.length === 0 ? (
   348→            <div className="flex flex-col items-center justify-center py-16 text-center">
   349→              <Users className="h-12 w-12 text-muted-foreground/30 mb-3" />
   350→              <p className="text-sm text-muted-foreground">No employees found</p>
   351→              <p className="text-xs text-muted-foreground mt-1">Try adjusting your search or filters</p>
   352→            </div>
   353→          ) : (
   354→            <div className="overflow-x-auto">
   355→              <Table>
   356→                <TableHeader>
   357→                  <TableRow className="hover:bg-transparent">
   358→                    <TableHead className="w-[280px]">Employee</TableHead>
   359→                    <TableHead>Role</TableHead>
   360→                    <TableHead className="hidden md:table-cell">Department</TableHead>
   361→                    <TableHead className="hidden lg:table-cell">Email</TableHead>
   362→                    <TableHead className="hidden lg:table-cell">Phone</TableHead>
   363→                    <TableHead className="w-[100px]">Status</TableHead>
   364→                  </TableRow>
   365→                </TableHeader>
   366→                <TableBody>
   367→                  {filtered.map((emp) => {
   368→                    const status = getStatus(emp);
   369→                    const statusMeta = STATUS_META[status] || STATUS_META.active;
   370→                    const dept = getDepartment(emp);

   381→                              <AvatarFallback
   382→                                className="text-xs font-bold"
   383→                                style={{
   384→                                  backgroundColor: NAVY,
   385→                                  color: "white",
   386→                                }}
   387→                              >
   388→                                {getInitials(emp.firstName, emp.lastName)}
   389→                              </AvatarFallback>
   390→                            </Avatar>
   391→                            <div className="min-w-0">
   392→                              <p className="text-sm font-semibold truncate">
   393→                                {emp.firstName} {emp.lastName}
   394→                              </p>
   395→                              <p className="text-[11px] text-muted-foreground truncate">

   398→                            </div>
   399→                          </div>

   440→// ─── TAB 2: DEPARTMENTS ────────────────────────────────────────────
   441→function DepartmentsTab() {
   442→  const { data: empData, loading } = useApi<any>("/api/hr/employees?isActive=all", []);
   443→
   444→  const employees: any[] = useMemo(() => {
   445→    if (!empData) return [];
   446→    return Array.isArray(empData) ? empData : [];
   447→  }, [empData]);
   448→
   449→  // Group by department
   450→  const deptGroups = useMemo(() => {
   451→    const map = new Map<string, any[]>();
   452→    employees.forEach((emp) => {
   453→      const dept = getDepartment(emp);
   454→      if (!map.has(dept)) map.set(dept, []);
   455→      map.get(dept)!.push(emp);
   456→    });
   457→    // Sort employees within each department by role level
   458→    map.forEach((emps) => {
   459→      emps.sort((a, b) => (ROLE_LEVEL[a.role] ?? 9) - (ROLE_LEVEL[b.role] ?? 9));
   460→    });
   461→    return Array.from(map.entries()).sort(([a], [b]) => a.localeCompare(b));
   462→  }, [employees]);
   463→
   464→  // Pie chart data
   465→  const pieData = useMemo(
   466→    () => deptGroups.map(([name, emps]) => ({ name, value: emps.length })),
   467→    [deptGroups]
   468→  );
   469→
   470→  const [selectedEmp, setSelectedEmp] = useState<any | null>(null);
   471→  const [detailOpen, setDetailOpen] = useState(false);
   472→
   473→  const openDetail = (emp: any) => {
   474→    setSelectedEmp(emp);
   475→    setDetailOpen(true);
   476→  };
   477→
   478→  if (loading) return <SkeletonGrid />;
   479→
   480→  return (
   481→    <div className="space-y-5">
   482→      {/* Department Distribution Chart */}
   483→      <div className="grid grid-cols-1 lg:grid-cols-2 gap-4">
   484→        <Card>
   485→          <CardHeader className="pb-2">
   486→            <CardTitle className="font-display text-sm">Department Distribution</CardTitle>
   487→          </CardHeader>
   488→          <CardContent>
   489→            {pieData.length > 0 ? (
   490→              <ResponsiveContainer width="100%" height={280}>
   491→                <PieChart>
   492→                  <Pie
   493→                    data={pieData}
   494→                    dataKey="value"
   495→                    nameKey="name"
   496→                    cx="50%"
   497→                    cy="50%"
   498→                    outerRadius={100}
   499→                    innerRadius={50}

   640→// ─── TAB 3: ORG CHART ──────────────────────────────────────────────
   641→function OrgChartTab() {
   642→  const { data: empData, loading } = useApi<any>("/api/hr/employees?isActive=all", []);
   643→
   644→  const employees: any[] = useMemo(() => {
   645→    if (!empData) return [];
   646→    return Array.isArray(empData) ? empData : [];
   647→  }, [empData]);
   648→
   649→  // Build hierarchy: group by role level, then by department
   650→  const hierarchy = useMemo(() => {
   651→    const levels = new Map<number, Map<string, any[]>>();
   652→
   653→    employees.forEach((emp) => {
   654→      const level = ROLE_LEVEL[emp.role] ?? 9;
   655→      const dept = getDepartment(emp);
   656→
   657→      if (!levels.has(level)) levels.set(level, new Map());
   658→      const deptMap = levels.get(level)!;
   659→      if (!deptMap.has(dept)) deptMap.set(dept, []);
   660→      deptMap.get(dept)!.push(emp);
   661→    });
   662→
   663→    // Sort each level's employees by name
   664→    levels.forEach((deptMap) => {
   665→      deptMap.forEach((emps) => {
   666→        emps.sort((a, b) => `${a.firstName} ${a.lastName}`.localeCompare(`${b.firstName} ${b.lastName}`));
   667→      });
   668→    });
   669→
   670→    return Array.from(levels.entries()).sort(([a], [b]) => a - b);
   671→  }, [employees]);
   672→
   673→  const LEVEL_LABELS: Record<number, string> = {
   674→    1: "Executive Leadership",
   675→    2: "General Management",
   676→    3: "Department Heads",
   677→    4: "Team Members",
   678→  };
   679→
   680→  const [selectedEmp, setSelectedEmp] = useState<any | null>(null);
   681→  const [detailOpen, setDetailOpen] = useState(false);
   682→  const [expandedDepts, setExpandedDepts] = useState<Set<string>>(new Set());
   683→
   684→  const toggleDept = (key: string) => {
   685→    setExpandedDepts((prev) => {
   686→      const next = new Set(prev);
   687→      if (next.has(key)) next.delete(key);
   688→      else next.add(key);
   689→      return next;
   690→    });
   691→  };
   692→
   693→  const openDetail = (emp: any) => {
   694→    setSelectedEmp(emp);
   695→    setDetailOpen(true);
   696→  };
   697→
   698→  if (loading) return <SkeletonGrid />;
   699→
   700→  return (
   701→    <div className="space-y-5">
   702→      {/* Org Overview KPIs */}
   703→      <div className="grid grid-cols-2 md:grid-cols-4 gap-3">
   704→        <KpiCard
   705→          label="Leadership"
   706→          value={employees.filter((e) => (ROLE_LEVEL[e.role] ?? 9) <= 2).length}
   707→          icon={GitBranch}
   708→          accent="navy"
   709→        />
   710→        <KpiCard
   711→          label="Dept Heads"
   712→          value={employees.filter((e) => (ROLE_LEVEL[e.role] ?? 9) === 3).length}
   713→          icon={Briefcase}
   714→          accent="gold"
   715→        />
   716→        <KpiCard
   717→          label="Team Members"
   718→          value={employees.filter((e) => (ROLE_LEVEL[e.role] ?? 9) >= 4).length}
   719→          icon={Users}
   720→          accent="success"
   721→        />
   722→        <KpiCard
   723→          label="Departments"
   724→          value={new Set(employees.map((e) => getDepartment(e))).size}
   725→          icon={Building2}
   726→          accent="info"
   727→        />
   728→      </div>
   729→
   730→      {/* Hierarchy Cards */}
   731→      <div className="space-y-6">
   732→        {hierarchy.map(([level, deptMap]) => {
   733→          const levelLabel = LEVEL_LABELS[level] || `Level ${level}`;
   734→          const isExecutive = level <= 2;
   735→          const isHead = level === 3;
   736→          const accentColor = isExecutive ? NAVY : isHead ? GOLD : SUCCESS;
   737→
   738→          return (
   739→            <div key={level}>
   740→              {/* Level Header */}
   741→              <div className="flex items-center gap-3 mb-3">
   742→                <div
   743→                  className="h-8 w-1.5 rounded-full"
   744→                  style={{ backgroundColor: accentColor }}
   745→                />
   746→                <h3 className="font-display text-base font-semibold" style={{ color: accentColor }}>
   747→                  {levelLabel}
   748→                </h3>
   749→                <div className="flex-1 border-b border-dashed" />
   750→                <Badge variant="outline" className="text-xs">
   751→                  {Array.from(deptMap.values()).reduce((sum, emps) => sum + emps.length, 0)} people
   752→                </Badge>
   753→              </div>
   754→
   755→              {/* Executive / GM — special centered layout */}
   756→              {isExecutive ? (
   757→                <div className="flex flex-wrap justify-center gap-4">
   758→                  {Array.from(deptMap.values())
   759→                    .flat()
   760→                    .map((emp) => (
   761→                      <OrgPersonCard
   762→                        key={emp.id}
   763→                        emp={emp}
   764→                        accentColor={accentColor}
   765→                        large
   766→                        onClick={() => openDetail(emp)}
   767→                      />
   768→                    ))}
   769→                </div>
   770→              ) : (
   771→                /* Department Heads & Members — grouped by department */
   772→                <div className="space-y-3 ml-4">
   773→                  {Array.from(deptMap.entries()).map(([dept, emps]) => {
   774→                    const deptKey = `${level}-${dept}`;
   775→                    const isExpanded = expandedDepts.has(deptKey);
   776→                    const deptColor = CHART_COLORS[
   777→                      Array.from(
   778→                        new Set(employees.map((e) => getDepartment(e)))
   779→                      ).indexOf(dept) % CHART_COLORS.length
   780→                    ] || SUCCESS;
   781→
   782→                    return (
   783→                      <Card key={dept} className="overflow-hidden">
   784→                        <div
   785→                          className="flex items-center gap-3 p-3 cursor-pointer hover:bg-muted/30 transition-colors"
   786→                          onClick={() => toggleDept(deptKey)}
   787→                        >
   788→                          {isExpanded ? (
   789→                            <ChevronDown className="h-4 w-4 text-muted-foreground shrink-0" />
   790→                          ) : (
   791→                            <ChevronRight className="h-4 w-4 text-muted-foreground shrink-0" />
   792→                          )}
   793→                          <div
   794→                            className="h-6 w-1.5 rounded-full"
   795→                            style={{ backgroundColor: deptColor }}
   796→                          />
   797→                          <span className="text-sm font-semibold">{dept}</span>
   798→                          <Badge variant="outline" className="text-[10px] ml-auto">
   799→                            {emps.length}
   800→                          </Badge>
   801→                        </div>
   802→                        {isExpanded && (
   803→                          <CardContent className="pt-0 pb-3 px-3">
   804→                            <div className="flex flex-wrap gap-3 ml-6">
   805→                              {emps.map((emp) => (
   806→                                <OrgPersonCard
   807→                                  key={emp.id}
   808→                                  emp={emp}
   809→                                  accentColor={deptColor}
   810→                                  onClick={() => openDetail(emp)}
   811→                                />
   812→                              ))}
   813→                            </div>
   814→                          </CardContent>
   815→                        )}
   816→                      </Card>
   817→                    );
   818→                  })}
   819→                </div>
   820→              )}
   821→            </div>
   822→          );
   823→        })}
   824→      </div>
   825→
   826→      {/* Employee Detail Dialog */}
   827→      <EmployeeDetailDialog
   828→        employee={selectedEmp}
   829→        open={detailOpen}
   830→        onClose={() => setDetailOpen(false)}
   831→      />
   832→    </div>
   833→  );
   834→}
