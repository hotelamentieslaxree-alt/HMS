    27→export function DashboardModule() {
    28→  const { role } = useAppStore();
    29→
    30→  // Route to the correct dashboard based on role
    31→  const roleLevel = role.includes("owner") ? "owner" :
    32→    role === "gm" ? "gm" :
    33→    role === "fom" || role === "receptionist" ? "frontdesk" :
    34→    role === "hk_mgr" || role === "hk_attendant" ? "housekeeping" :
    35→    role === "fb_mgr" || role === "waiter" ? "fnb" :
    36→    role === "fin_mgr" ? "finance" :
    37→    role === "eng_mgr" || role === "technician" ? "engineering" :
    38→    role === "rev_mgr" ? "revenue" :
    39→    role === "hr_mgr" ? "hr" :
    40→    "gm"; // fallback
    41→
    42→  switch (roleLevel) {
    43→    case "owner": return <OwnerDashboard />;
    44→    case "gm": return <GMDashboard />;
    45→    case "frontdesk": return <FrontDeskDashboard />;
    46→    case "housekeeping": return <HousekeepingDashboard />;
    47→    case "fnb": return <FnBDashboard />;
    48→    case "finance": return <FinanceDashboard />;
    49→    case "engineering": return <EngineeringDashboard />;
    50→    case "revenue": return <RevenueDashboard />;
    51→    case "hr": return <HRDashboard />;
    52→    default: return <GMDashboard />;
    53→  }
    54→}

    56→// ─── Shared hook for dashboard data ─────────────────────────────
    57→function useDashboardData() {
    58→  const { data, loading, reload } = useApi<any>("/api/dashboard", []);
    59→  const { refreshTick } = useAppStore();
    60→  useEffect(() => {
    61→    if (refreshTick > 0) reload();
    62→  }, [refreshTick, reload]);
    63→  return { data, loading };
    64→}
