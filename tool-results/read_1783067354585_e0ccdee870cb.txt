    98→// ─── Custom Tooltip for Recharts ──────────────────────────────────────────────
    99→function ChartTooltip({ active, payload, label }: any) {
   100→  if (!active || !payload?.length) return null;
   101→  return (
   102→    <div className="rounded-lg border border-border bg-card px-3 py-2 shadow-card-lg text-xs">
   103→      {label && <p className="font-semibold text-foreground mb-1">{label}</p>}
   104→      {payload.map((p: any, i: number) => (
   105→        <p key={i} style={{ color: p.color }} className="flex items-center gap-1.5">
   106→          <span className="h-2 w-2 rounded-full" style={{ backgroundColor: p.color }} />
   107→          {p.name}: <span className="font-mono-num font-bold">{typeof p.value === "number" && p.value > 999 ? fmtINR(p.value) : p.value}</span>
   108→        </p>
   109→      ))}
   110→    </div>
   111→  );
   112→}

   114→// ─── Source Badge ─────────────────────────────────────────────────────────────
   115→function SourceBadge({ source }: { source: string }) {
   116→  const meta = SOURCE_META[source];
   117→  if (!meta) return <Badge variant="secondary" className="text-[10px]">{source}</Badge>;
   118→  const Icon = meta.icon;
   119→  return (
   120→    <span
   121→      className="inline-flex items-center gap-1 rounded-md border px-2 py-0.5 text-[10px] font-semibold"
   122→      style={{ color: meta.color, borderColor: meta.color + "40", backgroundColor: meta.bg }}
   123→    >
   124→      <Icon className="h-3 w-3" />
   125→      {meta.label}
   126→    </span>
   127→  );
   128→}

   130→// ─── Pipeline Column ──────────────────────────────────────────────────────────
   131→function PipelineColumn({
   132→  stage,
   133→  leads,
   134→  onMoveLead,
   135→}: {
   136→  stage: typeof PIPELINE_STAGES[number];
   137→  leads: Lead[];
   138→  onMoveLead: (id: string, newStatus: string) => void;
   139→}) {
   140→  const stageLeads = leads.filter((l) => l.status === stage.key);
   141→  const totalValue = stageLeads.reduce((s, l) => s + l.estimatedValue, 0);
   142→  const prevStage = STATUS_FLOW[STATUS_FLOW.indexOf(stage.key as any) - 1];
   143→  const nextStage = STATUS_FLOW[STATUS_FLOW.indexOf(stage.key as any) + 1];
   144→
   145→  return (
   146→    <div className="flex flex-col min-w-[260px] w-[260px] shrink-0">
   147→      {/* Stage header */}
   148→      <div className="flex items-center gap-2 mb-2 px-1">
   149→        <span className="h-2.5 w-2.5 rounded-full shrink-0" style={{ backgroundColor: stage.color }} />
   150→        <h3 className="text-xs font-bold uppercase tracking-wider text-foreground">{stage.label}</h3>
   151→        <span className="ml-auto text-[10px] font-mono-num font-bold px-1.5 py-0.5 rounded-full" style={{ backgroundColor: stage.bg, color: stage.color }}>
   152→          {stageLeads.length}
   153→        </span>
   154→      </div>
   155→      <div className="mb-2 px-1">
   156→        <p className="text-[10px] text-muted-foreground">Pipeline Value</p>
   157→        <p className="text-sm font-mono-num font-bold" style={{ color: stage.color }}>{fmtINR(totalValue)}</p>
   158→      </div>
   159→
   160→      {/* Lead cards */}
   161→      <div className="flex-1 space-y-2 overflow-y-auto max-h-[calc(100vh-340px)] pr-1 scrollbar-thin">
   162→        {stageLeads.length === 0 ? (
   163→          <div className="flex items-center justify-center h-20 rounded-lg border border-dashed border-border text-[11px] text-muted-foreground">
   164→            No leads
   165→          </div>
   166→        ) : (
   167→          stageLeads.map((lead) => (
   168→            <Card key={lead.id} className="group shadow-card hover:shadow-card-lg transition-all cursor-default">
   169→              <CardContent className="p-3">
   170→                <div className="flex items-start justify-between gap-1 mb-1.5">
   171→                  <div className="min-w-0">
   172→                    <p className="text-xs font-bold text-foreground truncate">{lead.company}</p>
   173→                    <p className="text-[10px] text-muted-foreground flex items-center gap-1">
   174→                      <UserCircle className="h-3 w-3" /> {lead.contact}
   175→                    </p>
   176→                  </div>
   177→                  <span className="text-[10px] font-mono-num font-bold px-1.5 py-0.5 rounded" style={{ backgroundColor: stage.bg, color: stage.color }}>
   178→                    {lead.probability}%
   179→                  </span>
   180→                </div>
   181→
   182→                <div className="flex items-center justify-between mb-2">
   183→                  <span className="text-xs font-mono-num font-bold" style={{ color: NAVY }}>{fmtINR(lead.estimatedValue)}</span>
   184→                  <SourceBadge source={lead.source} />
   185→                </div>
   186→
   187→                <div className="flex items-center justify-between text-[10px] text-muted-foreground mb-2">
   188→                  <span className="flex items-center gap-0.5"><Clock className="h-3 w-3" />{timeAgo(lead.lastContacted)}</span>
   189→                  <span>{lead.assignedTo.split(" ")[0]}</span>
   190→                </div>
   191→
   192→                {/* Move buttons */}
   193→                <div className="flex items-center gap-1 opacity-0 group-hover:opacity-100 transition-opacity">
   194→                  {prevStage && stage.key !== "won" && stage.key !== "lost" && (
   195→                    <Button
   196→                      size="sm"
   197→                      variant="outline"
   198→                      className="h-6 text-[10px] px-2 flex-1"
   199→                      onClick={() => onMoveLead(lead.id, prevStage)}
   200→                    >
   201→                      <ChevronLeft className="h-3 w-3 mr-0.5" />
   202→                      {PIPELINE_STAGES.find((s) => s.key === prevStage)?.label}
   203→                    </Button>
   204→                  )}
   205→                  {nextStage && stage.key !== "won" && stage.key !== "lost" && (
   206→                    <Button
   207→                      size="sm"
   208→                      variant="outline"
   209→                      className="h-6 text-[10px] px-2 flex-1"
   210→                      onClick={() => onMoveLead(lead.id, nextStage)}
   211→                    >
   212→                      {PIPELINE_STAGES.find((s) => s.key === nextStage)?.label}
   213→                      <ChevronRight className="h-3 w-3 ml-0.5" />
   214→                    </Button>
   215→                  )}
   216→                  {stage.key !== "won" && stage.key !== "lost" && (
   217→                    <>
   218→                      <Button
   219→                        size="sm"
   220→                        className="h-6 text-[10px] px-2 bg-[#16A34A] hover:bg-[#15803D] text-white"
   221→                        onClick={() => onMoveLead(lead.id, "won")}
   222→                      >
   223→                        <Trophy className="h-3 w-3" />
   224→                      </Button>
   225→                      <Button
   226→                        size="sm"
   227→                        variant="outline"
   228→                        className="h-6 text-[10px] px-2 text-[#DC2626] border-[#DC2626]/40 hover:bg-[#FFE4E6]"

   231→                        <XCircle className="h-3 w-3" />
   232→                      </Button>
   233→                    </>

   235→                </div>
   236→              </CardContent>
   237→            </Card>
   238→          ))
   239→        )}
   240→      </div>

   243→}
