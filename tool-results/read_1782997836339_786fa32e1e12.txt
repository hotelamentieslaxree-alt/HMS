    91→// ─── TAB 1: OVERVIEW ───────────────────────────────────────────────
    92→function OverviewTab() {
    93→  const { data: empData, loading: l1 } = useApi<any>("/api/hr/employees?isActive=all", []);
    94→  const { data: attData, loading: l2 } = useApi<any>(`/api/hr/attendance?month=${now.getMonth()+1}&year=${now.getFullYear()}`, []);
    95→  const { data: scoreData, loading: l3 } = useApi<any>(`/api/hr/scorecards?period=${now.getFullYear()}-${String(now.getMonth()+1).padStart(2,"0")}`, []);
    96→  const { data: eventData, loading: l4 } = useApi<any>(`/api/hr/events?month=${now.getMonth()+1}&year=${now.getFullYear()}`, []);
    97→
    98→  if (l1 || l2 || l3 || l4 || !empData) return <SkeletonGrid />;
    99→
   100→  const emps: any[] = Array.isArray(empData) ? empData : [];
   101→  const totalEmployees = emps.length;
   102→  const activeCount = emps.filter((e: any) => e.isActive).length;
   103→  const byDepartment: Record<string, number> = {};
   104→  for (const e of emps) {
   105→    const d = e.department || "Unassigned";
   106→    byDepartment[d] = (byDepartment[d] || 0) + 1;
   107→  }
   108→  const attSummary = (attData as any).summary || {};
   109→  const overallStats = (scoreData as any).overallStats || {};
   110→  const events = (eventData as any).events || [];
   111→  const scorecards = (scoreData as any).scorecards || [];
   112→  const byDept = byDepartment;
   113→
   114→  // Attendance trend (last 7 days)
   115→  const attRecords = (attData as any).attendance || [];
   116→  const trend: { name: string; Present: number; Absent: number }[] = [];
   117→  for (let i = 6; i >= 0; i--) {
   118→    const d = new Date(); d.setDate(d.getDate() - i);
   119→    const ds = d.toISOString().slice(0, 10);
   120→    const dayRecs = attRecords.filter((r: any) => new Date(r.date).toISOString().slice(0, 10) === ds);
   121→    trend.push({
   122→      name: d.toLocaleDateString("en-IN", { weekday: "short" }),
   123→      Present: dayRecs.filter((r: any) => r.status === "present" || r.status === "late").length,
   124→      Absent: dayRecs.filter((r: any) => r.status === "absent").length,
   125→    });
   126→  }
   127→
   128→  const deptPie = Object.entries(byDept).map(([name, value]) => ({ name, value: value as number }));
   129→
   130→  // Estimate monthly payroll from active employees
   131→  const avgSalary = 45000;
   132→  const estPayroll = activeCount * avgSalary;
   133→
   134→  return (
   135→    <div className="space-y-5">
   136→      <div className="grid grid-cols-2 md:grid-cols-3 xl:grid-cols-6 gap-3">
   137→        <KpiCard label="Total Staff" value={totalEmployees} icon={Users} accent="navy" />
   138→        <KpiCard label="Active Today" value={activeCount} icon={UserCheck} accent="success" />
   139→        <KpiCard label="Attendance Rate" value={attSummary.attendanceRate ?? 0} unit="%" icon={Percent} accent="gold" />
   140→        <KpiCard label="Avg Score" value={overallStats.averageScore ?? 0} icon={Star} accent="info" />
   141→        <KpiCard label="Open Positions" value={3} icon={Briefcase} accent="warning" hint="FOM, Chef, HK" />
   142→        <KpiCard label="Monthly Payroll" value={fmtINR(estPayroll)} icon={IndianRupee} accent="error" />
   143→      </div>
   144→
   145→      <div className="grid grid-cols-1 lg:grid-cols-2 gap-4">
   146→        <Card>
   147→          <CardHeader className="pb-2"><CardTitle className="font-display text-sm">Attendance Trend (7 days)</CardTitle></CardHeader>
   148→          <CardContent>
   149→            <ResponsiveContainer width="100%" height={220}>
   150→              <BarChart data={trend}><CartesianGrid strokeDasharray="3 3" /><XAxis dataKey="name" tick={{ fontSize: 11 }} />
   151→                <YAxis tick={{ fontSize: 11 }} /><Tooltip /><Legend />
   152→                <Bar dataKey="Present" fill={CC[0]} radius={[4,4,0,0]} /><Bar dataKey="Absent" fill={CC[4]} radius={[4,4,0,0]} />
   153→              </BarChart>
   154→            </ResponsiveContainer>
   155→          </CardContent>
   156→        </Card>
   157→        <Card>
   158→          <CardHeader className="pb-2"><CardTitle className="font-display text-sm">Department Distribution</CardTitle></CardHeader>
   159→          <CardContent>
   160→            {deptPie.length > 0 ? (
   161→              <ResponsiveContainer width="100%" height={220}>
   162→                <PieChart><Pie data={deptPie} dataKey="value" nameKey="name" cx="50%" cy="50%" outerRadius={80} label={({ name, percent }) => `${name} ${(percent*100).toFixed(0)}%`} labelLine={false}>
   163→                  {deptPie.map((_, i) => <Cell key={i} fill={CC[i % CC.length]} />)}
   164→                </Pie><Tooltip /></PieChart>
   165→              </ResponsiveContainer>
   166→            ) : <p className="text-sm text-muted-foreground text-center py-10">No department data</p>}
   167→          </CardContent>
   168→        </Card>
   169→      </div>
   170→
   171→      <div className="grid grid-cols-1 lg:grid-cols-2 gap-4">
   172→        <Card>
   173→          <CardHeader className="pb-2"><CardTitle className="font-display text-sm">Recent Events</CardTitle></CardHeader>
   174→          <CardContent className="max-h-64 overflow-y-auto">
   175→            {events.length === 0 ? <p className="text-sm text-muted-foreground">No events this month</p> :
   176→              events.slice(0, 8).map((e: any) => (
   177→                <div key={e.id} className="flex items-center gap-3 py-2 border-b last:border-0">
   178→                  <Badge className={cn("text-[10px]", EVENT_COLORS[e.type] || "bg-muted")}>{e.type}</Badge>
   179→                  <div className="min-w-0 flex-1"><p className="text-sm font-medium truncate">{e.title}</p><p className="text-xs text-muted-foreground">{fmtDate(e.eventDate)}</p></div>
   180→                </div>
   181→              ))}
   182→          </CardContent>
   183→        </Card>
   184→        <Card>
   185→          <CardHeader className="pb-2"><CardTitle className="font-display text-sm">Top Performers</CardTitle></CardHeader>
   186→          <CardContent className="max-h-64 overflow-y-auto">
   187→            {scorecards.length === 0 ? <p className="text-sm text-muted-foreground">No scorecard data</p> :
   188→              scorecards.slice(0, 6).map((sc: any) => (
   189→                <div key={sc.id} className="flex items-center gap-3 py-2 border-b last:border-0">
   190→                  <div className="flex h-8 w-8 items-center justify-center rounded-full bg-[#1B3A6B] text-white text-xs font-bold">{(sc.userName||"?")[0]}</div>
   191→                  <div className="min-w-0 flex-1"><p className="text-sm font-medium truncate">{sc.userName}</p><p className="text-xs text-muted-foreground">{sc.department || sc.departmentCode || "—"}</p></div>
   192→                  <Badge className={cn("text-[10px]", GRADE_COLORS[sc.grade] || "")}>{sc.grade} — {sc.overallScore}</Badge>
   193→                </div>
   194→              ))}
   195→          </CardContent>
   196→        </Card>

   198→    </div>

   203→function AttendanceTab() {
   204→  const [month, setMonth] = useState(String(now.getMonth() + 1));
   205→  const [year, setYear] = useState(String(now.getFullYear()));
   206→  const [view, setView] = useState("monthly");
   207→  const [markOpen, setMarkOpen] = useState(false);
   208→  const [form, setForm] = useState({ userId: "", date: "", checkIn: "", checkOut: "", status: "present" });
   209→  const { data: empData } = useApi<any>("/api/hr/employees?isActive=true", []);
   210→  const { data, loading, reload } = useApi<any>(`/api/hr/attendance?month=${month}&year=${year}&view=${view}`, [month, year, view]);
   211→  const { triggerRefresh } = useAppStore();
   212→
   213→  const employees: any[] = Array.isArray(empData) ? empData : [];
   214→  const summary = data?.summary || {};
   215→  const records: any[] = data?.attendance || data?.todayAttendance || [];
   216→
   217→  const handleMark = async () => {
   218→    try {
   219→      await apiPost("/api/hr/attendance", { ...form, source: "manual" });
   220→      toast.success("Attendance recorded"); setMarkOpen(false); reload(); triggerRefresh();
   221→    } catch (e: any) { toast.error(e.message); }
   222→  };
   223→
   224→  const markAllPresent = async () => {
   225→    try {
   226→      const recs = employees.map((e: any) => ({ userId: e.id, date: new Date().toISOString().slice(0, 10), status: "present" as const }));
   227→      await apiPut("/api/hr/attendance", { records: recs });
   228→      toast.success(`Marked ${recs.length} present`); reload(); triggerRefresh();
   229→    } catch (e: any) { toast.error(e.message); }
   230→  };
   231→
   232→  if (loading || !data) return <SkeletonGrid />;
   233→
   234→  return (
   235→    <div className="space-y-4">
   236→      <div className="flex flex-wrap items-center gap-2">
   237→        <Select value={month} onValueChange={setMonth}>
   238→          <SelectTrigger className="w-28"><SelectValue /></SelectTrigger>
   239→          <SelectContent>{MONTHS.map((m, i) => <SelectItem key={i} value={String(i+1)}>{m}</SelectItem>)}</SelectContent>
   240→        </Select>
   241→        <Input className="w-24" value={year} onChange={(e) => setYear(e.target.value)} placeholder="Year" />
   242→        <div className="flex gap-1">
   243→          {["daily","weekly","monthly"].map(v => (
   244→            <Button key={v} size="sm" variant={view === v ? "default" : "outline"} onClick={() => setView(v)}>{v.charAt(0).toUpperCase()+v.slice(1)}</Button>
   245→          ))}
   246→        </div>
   247→        <div className="ml-auto flex gap-2">
   248→          <Button size="sm" variant="outline" onClick={markAllPresent}><CheckCheck className="h-3.5 w-3.5 mr-1" />Mark All Present</Button>
   249→          <Button size="sm" variant="outline" onClick={() => toast.info("Excel upload coming soon")}><Upload className="h-3.5 w-3.5 mr-1" />Upload Excel</Button>
   250→          <Button size="sm" onClick={() => setMarkOpen(true)}><Plus className="h-3.5 w-3.5 mr-1" />Mark Attendance</Button>
   251→        </div>
   252→      </div>
   253→
   254→      <div className="grid grid-cols-2 md:grid-cols-3 xl:grid-cols-6 gap-3">
   255→        <KpiCard label="Present" value={summary.totalPresent ?? 0} icon={UserCheck} accent="success" />
   256→        <KpiCard label="Absent" value={summary.totalAbsent ?? 0} icon={Users} accent="error" />
   257→        <KpiCard label="Late" value={summary.totalLate ?? 0} icon={Clock} accent="warning" />
   258→        <KpiCard label="Half Day" value={summary.totalHalfDay ?? 0} icon={Percent} accent="info" />
   259→        <KpiCard label="On Leave" value={summary.totalOnLeave ?? 0} icon={CalendarDays} accent="navy" />
   260→        <KpiCard label="Att. Rate" value={summary.attendanceRate ?? 0} unit="%" icon={BarChart3} accent="gold" />
   261→      </div>
   262→
   263→      <div className="overflow-x-auto rounded-lg border">
   264→        <Table>
   265→          <TableHeader><TableRow>
   266→            <TableHead>Employee</TableHead><TableHead>Date</TableHead><TableHead>Check-In</TableHead><TableHead>Check-Out</TableHead>
   267→            <TableHead>Hours</TableHead><TableHead>Overtime</TableHead><TableHead>Status</TableHead><TableHead>Source</TableHead>
   268→          </TableRow></TableHeader>
   269→          <TableBody>
   270→            {records.length === 0 ? (
   271→              <TableRow><TableCell colSpan={8} className="text-center text-muted-foreground py-8">No attendance records</TableCell></TableRow>

   273→              <TableRow key={r.id}>
   274→                <TableCell className="font-medium">{r.user?.firstName} {r.user?.lastName}<p className="text-[10px] text-muted-foreground">{r.user?.employeeCode}</p></TableCell>

   279→                <TableCell className="font-mono-num text-xs">{r.overtimeHours || "—"}</TableCell>

   281→                <TableCell className="text-xs capitalize">{r.source}</TableCell>

   308→    </div>

   312→// ─── TAB 3: EMPLOYEES ──────────────────────────────────────────────
   313→function EmployeesTab() {
   314→  const [search, setSearch] = useState("");
   315→  const [deptFilter, setDeptFilter] = useState("all");
   316→  const [addOpen, setAddOpen] = useState(false);
   317→  const [editItem, setEditItem] = useState<any>(null);
   318→  const [form, setForm] = useState({ firstName: "", lastName: "", email: "", role: "", department: "", phone: "" });
   319→  const { data, loading, reload } = useApi<any>(`/api/hr/employees?isActive=all${search ? `&search=${search}` : ""}${deptFilter !== "all" ? `&department=${deptFilter}` : ""}`, [search, deptFilter]);
   320→  const { triggerRefresh } = useAppStore();
   321→
   322→  const employees: any[] = Array.isArray(data) ? data : [];
   323→  const summary = (data as any)?.summary || {};
   324→  const departments = Object.keys(summary.byDepartment || {});
   325→
   326→  const resetForm = () => setForm({ firstName: "", lastName: "", email: "", role: "", department: "", phone: "" });
   327→
   328→  const handleSave = async () => {
   329→    try {
   330→      const body = { ...form, departmentId: form.department || undefined };
   331→      if (editItem) { await apiPut("/api/hr/employees", { id: editItem.id, ...body }); toast.success("Employee updated"); }
   332→      else { await apiPost("/api/hr/employees", body); toast.success("Employee added"); }
   333→      setAddOpen(false); setEditItem(null); resetForm(); reload(); triggerRefresh();
   334→    } catch (e: any) { toast.error(e.message); }
   335→  };
   336→
   337→  const handleDelete = async (id: string) => {
   338→    try { await apiPut("/api/hr/employees", { id, isActive: false }); toast.success("Employee deactivated"); reload(); triggerRefresh(); }
   339→    catch (e: any) { toast.error(e.message); }
   340→  };
   341→
   342→  if (loading || !data) return <SkeletonGrid />;
   343→
   344→  return (
   345→    <div className="space-y-4">
   346→      <div className="flex flex-wrap items-center gap-2">
   347→        <div className="relative flex-1 min-w-[200px]"><Search className="absolute left-2.5 top-2.5 h-4 w-4 text-muted-foreground" /><Input className="pl-9" placeholder="Search employees..." value={search} onChange={e => setSearch(e.target.value)} /></div>
   348→        <Select value={deptFilter} onValueChange={setDeptFilter}>
   349→          <SelectTrigger className="w-44"><SelectValue placeholder="Department" /></SelectTrigger>
   350→          <SelectContent><SelectItem value="all">All Departments</SelectItem>{departments.map(d => <SelectItem key={d} value={d}>{d}</SelectItem>)}</SelectContent>
   351→        </Select>
   352→        <Button size="sm" onClick={() => { resetForm(); setEditItem(null); setAddOpen(true); }}><Plus className="h-3.5 w-3.5 mr-1" />Add Employee</Button>
   353→      </div>
   354→

   697→            {[
