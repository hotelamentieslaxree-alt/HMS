   104→const now = new Date();

   106→function currentPeriod() {
   107→  return `${now.getFullYear()}-${String(now.getMonth() + 1).padStart(2, "0")}`;
   108→}
   109→
   110→function periodLabel(p: string) {
   111→  const [y, m] = p.split("-");
   112→  const months = [
   113→    "Jan","Feb","Mar","Apr","May","Jun",
   114→    "Jul","Aug","Sep","Oct","Nov","Dec",
   115→  ];
   116→  return `${months[Number(m) - 1]} ${y}`;
   117→}
   118→
   119→function last12Periods(): string[] {
   120→  const periods: string[] = [];
   121→  for (let i = 0; i < 12; i++) {
   122→    const d = new Date(now.getFullYear(), now.getMonth() - i, 1);
   123→    periods.push(
   124→      `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}`
   125→    );
   126→  }
   127→  return periods;
   128→}

   517→// ─── Scorecards Table Tab ─────────────────────────────────────────
   518→function ScorecardsTab({
   519→  onViewEmployee,
   520→}: {
   521→  onViewEmployee: (userId: string, userName: string) => void;
   522→}) {
   523→  const [period, setPeriod] = useState(currentPeriod());
   524→  const [search, setSearch] = useState("");
   525→  const [deptFilter, setDeptFilter] = useState("all");
   526→  const [addOpen, setAddOpen] = useState(false);
   527→  const [editSc, setEditSc] = useState<ScorecardRow | null>(null);
   528→
   529→  const { data, loading, reload } = useApi<ScorecardData>(
   530→    `/api/hr/scorecards?period=${period}`,
   531→    [period]
   532→  );
   533→  const { data: empData } = useApi<Employee[]>(
   534→    "/api/hr/employees?isActive=true",
   535→    []
   536→  );
   537→  const { triggerRefresh } = useAppStore();
   538→
   539→  const employees: Employee[] = Array.isArray(empData) ? empData : [];
   540→  const scorecards: ScorecardRow[] = data?.scorecards || [];
   541→  const departments = useMemo(() => {
   542→    const deptSet = new Set<string>();
   543→    scorecards.forEach((sc) => {
   544→      if (sc.department) deptSet.add(sc.department);
   545→    });
   546→    return Array.from(deptSet).sort();
   547→  }, [scorecards]);
   548→
   549→  const filtered = useMemo(() => {
   550→    let result = scorecards;
   551→    if (search) {
   552→      const q = search.toLowerCase();
   553→      result = result.filter(
   554→        (sc) =>
   555→          sc.userName?.toLowerCase().includes(q) ||
   556→          sc.department?.toLowerCase().includes(q)
   557→      );
   558→    }
   559→    if (deptFilter !== "all") {
   560→      result = result.filter((sc) => sc.department === deptFilter);
   561→    }
   562→    return result;
   563→  }, [scorecards, search, deptFilter]);
   564→
   565→  const handleSave = async (metrics: MetricFormState) => {
   566→    try {
   567→      if (editSc) {
   568→        await apiPut("/api/hr/scorecards", {
   569→          id: editSc.id,
   570→          ...metrics,
   571→          period,
   572→        });
   573→        toast.success("Scorecard updated");
   574→      } else {
   575→        await apiPost("/api/hr/scorecards", { ...metrics, period });
   576→        toast.success("Scorecard saved");
   577→      }
   578→      setAddOpen(false);
   579→      setEditSc(null);
   580→      reload();
   581→      triggerRefresh();
   582→    } catch (e: any) {
   583→      toast.error(e.message);
   584→    }
   585→  };
   586→
   587→  if (loading || !data) return <SkeletonOverview />;
   588→
   589→  return (
   590→    <div className="space-y-4">
   591→      {/* Filters */}
   592→      <div className="flex flex-wrap items-center gap-2">
   593→        <Select value={period} onValueChange={setPeriod}>
   594→          <SelectTrigger className="w-40">
   595→            <SelectValue />
   596→          </SelectTrigger>
   597→          <SelectContent>
   598→            {last12Periods().map((p) => (
   599→              <SelectItem key={p} value={p}>
   600→                {periodLabel(p)}
   601→              </SelectItem>
   602→            ))}
   603→          </SelectContent>
   604→        </Select>
   605→
   606→        <div className="relative">
   607→          <Search className="absolute left-2.5 top-1/2 h-3.5 w-3.5 -translate-y-1/2 text-muted-foreground" />
   608→          <Input
   609→            className="w-52 pl-8"
   610→            placeholder="Search employee…"
   611→            value={search}
   612→            onChange={(e) => setSearch(e.target.value)}
   613→          />
   614→        </div>
   615→
   616→        <Select value={deptFilter} onValueChange={setDeptFilter}>
   617→          <SelectTrigger className="w-40">
   618→            <SelectValue />
   619→          </SelectTrigger>
   620→          <SelectContent>
   621→            <SelectItem value="all">All Departments</SelectItem>
   622→            {departments.map((d) => (
   623→              <SelectItem key={d} value={d}>
   624→                {d}
